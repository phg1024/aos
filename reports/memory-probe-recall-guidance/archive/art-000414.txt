//
// One thread owns one pixel and traces `params.spp` camera samples in order,
// accumulating into a private float3. Keeping the sample loop inside the
// thread makes the per-pixel sum a fixed-order float add, so a given seed
// reproduces the PFM bit for bit. The host prepends <metal_stdlib> and
// pt_gpu_types.hpp before compiling, so only kernel code appears here.

// ---------------------------------------------------------------------------
// Deterministic RNG (PCG-XSH) and small helpers.
// ---------------------------------------------------------------------------

inline uint pt_pcg_next(thread uint& state) {
    state = state * 747796405u + 2891336453u;
    const uint word = ((state >> ((state >> 28u) + 4u)) ^ state) * 277803737u;
    return (word >> 22u) ^ word;
}

inline float pt_random(thread uint& state) {
    return float(pt_pcg_next(state)) * (1.0 / 4294967296.0);
}

// Per-pixel, per-sample stream seed: a pure function of (seed, pixel, sample).
inline uint pt_seed_pixel(uint seed, uint pixel, uint sample) {
    uint state = seed * 747796405u + pixel * 2654435761u + sample * 1597334677u;
    return pt_pcg_next(state);
}

inline float3 pt_v3(device const float* a) {
    return float3(a[0], a[1], a[2]);
}

inline float3 pt_orthonormal_tangent(float3 n) {
    // Duff et al. branchless tangent frame construction.
    const float sign = copysign(1.0f, n.z);
    const float a = -1.0f / (sign + n.z);
    const float b = n.x * n.y * a;
    return float3(1.0f + sign * a * n.x * n.x, sign * b, -sign * n.x);
}

struct PtRay {
    float3 origin;
    float3 direction;
    float3 inv_direction;
};

inline PtRay pt_make_ray(float3 origin, float3 direction) {
    PtRay ray;
    ray.origin = origin;
    ray.direction = direction;
    ray.inv_direction = float3(1.0f / direction.x, 1.0f / direction.y, 1.0f / direction.z);
    return ray;
}

enum PtMaterial {
    PtMaterialNone = 0,
    PtMaterialFloor = 1,
    PtMaterialLight = 2,
    PtMaterialGlass = 3
};

struct PtHit {
    float t;
    uint material;
    float3 normal;
    float3 position;
};

struct PtScene {
    device const PtParams* params;
    device const PtVertex* vertices;
    device const uint* triangles;
    device const PtBvhNode* nodes;
    device const uint* order;
};

// Ray/axis-aligned box slab test using the precomputed inverse direction.
inline bool pt_ray_aabb(device const PtBvhNode& node, float3 origin, float3 inv_direction,
                        float t_min, float t_max) {
    const float3 bmin = pt_v3(node.bmin);
    const float3 bmax = pt_v3(node.bmax);
    const float3 t0 = (bmin - origin) * inv_direction;
    const float3 t1 = (bmax - origin) * inv_direction;
    const float3 lower = fmin(t0, t1);
    const float3 upper = fmax(t0, t1);
    const float entered = fmax(fmax(lower.x, lower.y), max(lower.z, t_min));
    const float exited = fmin(fmin(upper.x, upper.y), min(upper.z, t_max));
    return entered <= exited && exited > 0.0f;
}

// Moller-Trumbore, double sided because the glass shows backfaces.
inline bool pt_ray_triangle(float3 origin, float3 direction, float3 e1, float3 e2, float3 v0,
                            thread float& t, thread float& u, thread float& v) {
    const float3 p = cross(direction, e2);
    const float det = dot(e1, p);
    if (fabs(det) < 1e-12f) {
        return false;
    }
    const float inv_det = 1.0f / det;
    const float3 s = origin - v0;
    u = dot(s, p) * inv_det;
    if (u < 0.0f || u > 1.0f) {
        return false;
    }
    const float3 q = cross(s, e1);
    v = dot(direction, q) * inv_det;
    if (v < 0.0f || u + v > 1.0f) {
        return false;
    }
    t = dot(e2, q) * inv_det;
    return t > 1e-5f;
}

inline bool pt_ray_plane_y(float3 origin, float3 direction, float height, thread float& t) {
    if (fabs(direction.y) < 1e-9f) {
        return false;
    }
    t = (height - origin.y) / direction.y;
    return t > 1e-4f;
}

// Rectangle spanned by light_u/light_v around light_center, seen only from the
// emission side (a one-sided emitter).
inline bool pt_ray_light_rect(device const PtParams& params, float3 origin, float3 direction,
                              thread float& t, thread float3& point) {
    const float3 normal = pt_v3(params.light_normal);
    const float denom = dot(direction, normal);
    if (denom > -1e-9f) {
        return false;
    }
    const float3 center = pt_v3(params.light_center);
    const float candidate = dot(center - origin, normal) / denom;
    if (candidate <= 1e-4f) {
        return false;
    }
    const float3 hit_point = origin + candidate * direction;
    const float3 delta = hit_point - center;
    const float su = dot(delta, pt_v3(params.light_u));
    const float sv = dot(delta, pt_v3(params.light_v));
    if (fabs(su) > params.light_half_u || fabs(sv) > params.light_half_v) {
        return false;
    }
    t = candidate;
    point = hit_point;
    return true;
}

// BVH traversal with a fixed per-thread stack; nearest-hit only.
inline bool pt_intersect_mesh(PtScene scene, PtRay ray, thread float& best_t,
                              thread float3& best_normal) {    uint stack[PT_BVH_STACK_SIZE];
    int stack_size = 0;
    uint node_index = 0;
    bool found = false;

    while (true) {
        device const PtBvhNode& node = scene.nodes[node_index];
        if (pt_ray_aabb(node, ray.origin, ray.inv_direction, 1e-4f, best_t)) {
            if (node.count > 0u) {
                for (uint i = 0; i < node.count; ++i) {
                    const uint tri = scene.order[node.start + i] * 3u;
                    const uint i0 = scene.triangles[tri + 0];
                    const uint i1 = scene.triangles[tri + 1];
                    const uint i2 = scene.triangles[tri + 2];
                    const float3 v0 = pt_v3(scene.vertices[i0].position);
                    const float3 v1 = pt_v3(scene.vertices[i1].position);
                    const float3 v2 = pt_v3(scene.vertices[i2].position);
                    float t = 0.0f;
                    float u = 0.0f;
                    float v = 0.0f;
                    if (pt_ray_triangle(ray.origin, ray.direction, v1 - v0, v2 - v0, v0, t, u, v)) {
                        if (t < best_t) {
                            best_t = t;
                            const float3 n0 = pt_v3(scene.vertices[i0].normal);
                            const float3 n1 = pt_v3(scene.vertices[i1].normal);
                            const float3 n2 = pt_v3(scene.vertices[i2].normal);
                            best_normal = normalize((1.0f - u - v) * n0 + u * n1 + v * n2);
                            found = true;
                        }
                    }
                }
            } else if (stack_size < PT_BVH_STACK_SIZE) {
                // Push the farther child and descend into the nearer one.
                stack[stack_size++] = node.right;
                node_index = node.left;
                continue;
            }
        }
        if (stack_size == 0) {
            break;
        }
        node_index = stack[--stack_size];
    }
    return found;
}

// `test_light` is false for shadow rays: the emitter must never occlude itself,
// otherwise every next-event sample would report a hit at the light's own
// surface. The light is one-sided, so a ray leaving it is not blocked by it.
inline PtHit pt_intersect_scene(PtScene scene, PtRay ray, bool test_light) {
    PtHit hit;
    hit.t = 1e30f;
    hit.material = PtMaterialNone;
    hit.normal = float3(0.0f, 1.0f, 0.0f);
    hit.position = float3(0.0f);

    float t_floor = 0.0f;
    if (pt_ray_plane_y(ray.origin, ray.direction, scene.params->floor_height, t_floor) &&
        t_floor < hit.t) {
        hit.t = t_floor;
        hit.material = PtMaterialFloor;
        hit.normal = float3(0.0f, 1.0f, 0.0f);
        hit.position = ray.origin + t_floor * ray.direction;
    }

    float t_light = 0.0f;
    float3 light_point = float3(0.0f);
    if (test_light && pt_ray_light_rect(*scene.params, ray.origin, ray.direction, t_light,
                                       light_point) && t_light < hit.t) {
        hit.t = t_light;
        hit.material = PtMaterialLight;
        hit.normal = pt_v3(scene.params->light_normal);
        hit.position = light_point;
    }

    float3 mesh_normal = float3(0.0f, 1.0f, 0.0f);
    float mesh_t = hit.t;
    if (pt_intersect_mesh(scene, ray, mesh_t, mesh_normal)) {
        hit.t = mesh_t;
        hit.material = PtMaterialGlass;
        hit.normal = mesh_normal;
        hit.position = ray.origin + mesh_t * ray.direction;
    }
    return hit;
}

// Shadow test: anything opaque before `t_limit` blocks the light. The emitter
// itself is excluded so a shadow ray can reach it.
inline bool pt_occluded(PtScene scene, PtRay ray, float t_limit) {
    const PtHit probe = pt_intersect_scene(scene, ray, false);
    return probe.material != PtMaterialNone && probe.t < t_limit - 1e-3f;
}

//
// Unbiased unidirectional path tracing. Diffuse floor vertices get next-event
// estimation against the rectangular area light in addition to BSDF sampling;
// glass vertices are pure BSDF sampling with a delta specular lobe, so their
// light hits are counted directly. Emission along a continued path is added
// only for the primary ray and after specular bounces, which is what keeps the
// NEE term from being double counted.

inline float3 pt_environment(device const PtParams& params) {
    return pt_v3(params.env_radiance);
}

// Schlick approximation of the Fresnel equations for a dielectric interface
// between ior_i (the side the ray comes from) and ior_t (the other side).
inline float pt_fresnel(float cos_incident, float ior_i, float ior_t) {
    const float c = clamp(cos_incident, 0.0f, 1.0f);
    const float eta = ior_t / ior_i;
    const float sin2_t = eta * eta * (1.0f - c * c);
    if (sin2_t > 1.0f) {
        return 1.0f;  // total internal reflection
    }
    const float r0 = (ior_i - ior_t) / (ior_i + ior_t);
    const float r0sq = r0 * r0;
    const float one_minus_c = 1.0f - c;
    const float f5 = one_minus_c * one_minus_c * one_minus_c * one_minus_c * one_minus_c;
    return r0sq + (1.0f - r0sq) * f5;
}

// Snell's law. `incident` is the unit ray direction, `surface_normal` is the
// unit normal facing against it, `eta` is n_i / n_t. False means TIR.
inline bool pt_refract(float3 incident, float3 surface_normal, float eta, thread float3& result) {
    const float cos_incident = -dot(incident, surface_normal);  // positive by construction
    const float sin2_t = eta * eta * (1.0f - cos_incident * cos_incident);
    if (sin2_t > 1.0f) {
        return false;
    }
    const float cos_t = sqrt(max(0.0f, 1.0f - sin2_t));
    result = normalize(eta * incident + (eta * cos_incident - cos_t) * surface_normal);
    return true;
}

// Uniform sample of the emitter rectangle; returns the sampled point.
inline float3 pt_sample_light(device const PtParams& params, thread uint& state) {
    const float3 center = pt_v3(params.light_center);
    const float3 u_axis = pt_v3(params.light_u);
    const float3 v_axis = pt_v3(params.light_v);
    const float su = (2.0f * pt_random(state) - 1.0f) * params.light_half_u;
    const float sv = (2.0f * pt_random(state) - 1.0f) * params.light_half_v;
    return center + su * u_axis + sv * v_axis;
}

// Cosine-weighted hemisphere direction around `normal`.
inline float3 pt_sample_cosine(float3 normal, thread uint& state) {
    const float r1 = pt_random(state);
    const float r2 = pt_random(state);
    const float phi = 6.283185307179586f * r2;
    const float sin_theta = sqrt(r1);
    const float cos_theta = sqrt(max(0.0f, 1.0f - r1));
    const float3 tangent = pt_orthonormal_tangent(normal);
    const float3 bitangent = cross(normal, tangent);
    return normalize(tangent * (cos(phi) * sin_theta) + bitangent * (sin(phi) * sin_theta) +
                         normal * cos_theta);
}

// One bounce of the integrator loop. Returns false when the path is finished.
inline bool pt_trace_path(PtScene scene, PtRay ray, uint pixel_seed, thread float3& radiance) {
    device const PtParams& params = *scene.params;
    uint state = pixel_seed;
    float3 throughput = float3(1.0f);
    bool inside_glass = false;
    // Whether the previous vertex was specular (mirror/glass). Emission at a
    // light hit is counted only for the primary ray and after a specular
    // bounce; diffuse vertices collect the light through next-event estimation.
    bool last_specular = true;

    for (uint depth = 0; depth < params.max_depth; ++depth) {
        const PtHit hit = pt_intersect_scene(scene, ray, true);

        if (hit.material == PtMaterialNone) {
            radiance += throughput * pt_environment(params);
            return false;
        }

        if (hit.material == PtMaterialLight) {
            if (depth == 0u || last_specular) {
                radiance += throughput * pt_v3(params.light_radiance);
            }
            return false;
        }

        if (hit.material == PtMaterialFloor) {
            const float3 normal = hit.normal;

            // Next-event estimation: uniform sample of the light rectangle.
            const float3 light_point = pt_sample_light(params, state);
            const float3 to_light = light_point - hit.position;
            const float distance_sq = max(1e-8f, dot(to_light, to_light));
            const float3 light_direction = to_light * rsqrt(distance_sq);
            const float incident_cosine = dot(normal, light_direction);
            // Cosine of the light's own lobe: emission direction toward x.
            const float light_cosine = -dot(pt_v3(params.light_normal), light_direction);
            if (incident_cosine > 0.0f && light_cosine > 0.0f) {
                const PtRay shadow_ray =
                    pt_make_ray(hit.position + normal * 1e-4f, light_direction);
                const float shadow_limit = sqrt(distance_sq);
                if (!pt_occluded(scene, shadow_ray, shadow_limit)) {
                    // Contribution = (albedo/pi) * Le * cos_i * cos_l * A / d^2:
                    // f_r times the geometry term divided by the area pdf 1/A.
                    const float weight = params.floor_albedo * incident_cosine * light_cosine *
                                         params.light_area / (3.14159265358979f * distance_sq);
                    radiance += throughput * weight * pt_v3(params.light_radiance);
                }
            }

            // Continue along a cosine-weighted diffuse direction. A diffuse
            // vertex ends any glass traversal and forbids counting emission
            // directly on the next vertex.
            throughput *= params.floor_albedo;
            inside_glass = false;
            last_specular = false;
            ray = pt_make_ray(hit.position + normal * 1e-4f, pt_sample_cosine(normal, state));
        } else {
            // Clear glass: sample reflection or refraction as a delta lobe.
            const bool entering = dot(ray.direction, hit.normal) < 0.0f;
            const float3 oriented_normal = entering ? hit.normal : -hit.normal;
            const float ior_i = inside_glass ? params.ior : 1.0f;
            const float ior_t = inside_glass ? 1.0f : params.ior;
            const float ratio = ior_t / ior_i;
            const float reflectance =
                pt_fresnel(-dot(ray.direction, oriented_normal), ior_i, ior_t);

            float3 reflected = reflect(ray.direction, oriented_normal);
            float3 transmitted = float3(0.0f);
            const bool refracted =
                pt_refract(ray.direction, oriented_normal, ior_i / ior_t, transmitted);

            bool do_reflect;
            if (!refracted) {
                do_reflect = true;  // total internal reflection
            } else {
                do_reflect = pt_random(state) < reflectance;
            }

            if (do_reflect) {
                // Sampling probability equals the Fresnel reflectance, so the
                // reflectance cancels and the throughput factor is 1.
                last_specular = true;
                ray = pt_make_ray(hit.position + oriented_normal * 1e-4f, reflected);
            } else {
                // Radiance scales by (n_t/n_i)^2 across a dielectric; the
                // (1-F) transmittance cancels the sampling probability.
                throughput *= ratio * ratio;
                ray = pt_make_ray(hit.position - oriented_normal * 1e-4f, transmitted);
                inside_glass = entering;
                last_specular = true;
            }
        }

        // Unbiased Russian roulette, only once the path is deep enough.
        if (depth >= 4u) {
            const float survival = clamp(max(throughput.x, max(throughput.y, throughput.z)),
                                         0.05f, 0.95f);
            if (pt_random(state) > survival) {
                return false;
            }
            throughput /= survival;
        }

#include <algorithm>
#include <cmath>

namespace pt {
namespace {

constexpr unsigned int kLeafSize = 4;

struct TriBounds {
    float bmin[3];
    float bmax[3];
    float centroid[3];
};

void triangle_bounds(const Mesh& mesh, unsigned int tri_index, TriBounds& out) {
    const PtVertex* v[3] = {&mesh.vertices[mesh.triangles[3 * tri_index + 0]],
                            &mesh.vertices[mesh.triangles[3 * tri_index + 1]],
                            &mesh.vertices[mesh.triangles[3 * tri_index + 2]]};
    for (int axis = 0; axis < 3; ++axis) {
        float lo = v[0]->position[axis];
        float hi = lo;
        for (int k = 1; k < 3; ++k) {
            lo = std::min(lo, v[k]->position[axis]);
            hi = std::max(hi, v[k]->position[axis]);
        }
        out.bmin[axis] = lo;
        out.bmax[axis] = hi;
        out.centroid[axis] = 0.5f * (lo + hi);
    }
}

void union_bounds(const TriBounds& src, float bmin[3], float bmax[3]) {
    for (int axis = 0; axis < 3; ++axis) {
        bmin[axis] = std::min(bmin[axis], src.bmin[axis]);
        bmax[axis] = std::max(bmax[axis], src.bmax[axis]);
    }
}

struct BuildFrame {
    unsigned int node;
    unsigned int begin;
    unsigned int end;
};

}  // namespace

void transform_mesh(Mesh& mesh, float scale, const float translate[3]) {
    for (PtVertex& v : mesh.vertices) {
        for (int axis = 0; axis < 3; ++axis) {
            v.position[axis] = scale * v.position[axis] + translate[axis];
        }
    }
}

void build_bvh(const Mesh& mesh, Bvh& out) {
    const size_t tri_count = mesh.triangles.size() / 3;
    out.triangle_order.resize(tri_count);
    for (size_t i = 0; i < tri_count; ++i) {
        out.triangle_order[i] = static_cast<unsigned int>(i);
    }

    std::vector<TriBounds> bounds(tri_count);
    for (size_t i = 0; i < tri_count; ++i) {
        triangle_bounds(mesh, static_cast<unsigned int>(i), bounds[i]);
    }

    out.nodes.clear();
    out.nodes.reserve(tri_count * 2);
    out.nodes.push_back(PtBvhNode());

    std::vector<BuildFrame> stack;
    stack.push_back(BuildFrame{0, 0, static_cast<unsigned int>(tri_count)});

    while (!stack.empty()) {
        const BuildFrame frame = stack.back();
        stack.pop_back();

        float bmin[3] = {1e30f, 1e30f, 1e30f};
        float bmax[3] = {-1e30f, -1e30f, -1e30f};
        for (unsigned int i = frame.begin; i < frame.end; ++i) {
            union_bounds(bounds[out.triangle_order[i]], bmin, bmax);
        }

        PtBvhNode& node = out.nodes[frame.node];
        for (int axis = 0; axis < 3; ++axis) {
            node.bmin[axis] = bmin[axis];
            node.bmax[axis] = bmax[axis];
        }

        const unsigned int count = frame.end - frame.begin;
        if (count <= kLeafSize) {
            node.count = count;
            node.start = frame.begin;
            node.left = 0;
            node.right = 0;
            continue;
        }

        // Split at the median centroid along the widest axis.
        float extent[3] = {bmax[0] - bmin[0], bmax[1] - bmin[1], bmax[2] - bmin[2]};
        int axis = 0;
        if (extent[1] > extent[axis]) {
            axis = 1;
        }
        if (extent[2] > extent[axis]) {
            axis = 2;
        }

        const unsigned int middle = frame.begin + count / 2;
        std::nth_element(out.triangle_order.begin() + frame.begin,
                         out.triangle_order.begin() + middle,
                         out.triangle_order.begin() + frame.end,
                         [axis, &bounds](unsigned int a, unsigned int b) {
                             return bounds[a].centroid[axis] < bounds[b].centroid[axis];
                         });

        const unsigned int left_index = static_cast<unsigned int>(out.nodes.size());
        out.nodes.push_back(PtBvhNode());
        out.nodes.push_back(PtBvhNode());
        node.count = 0;
        node.start = 0;
        node.left = left_index;
        node.right = left_index + 1;

        stack.push_back(BuildFrame{left_index + 1, middle, frame.end});
        stack.push_back(BuildFrame{left_index, frame.begin, middle});
    }
}

bool MetalPathTracer::init(const std::string& shader_dir, const std::string& shared_header_path,
                           std::string& error) {
    impl_->device = MTLCreateSystemDefaultDevice();
    if (impl_->device == nil) {
        return impl_->fail(@"no Metal device available", error);
    }
    impl_->queue = [impl_->device newCommandQueue];

    NSString* dir = [NSString stringWithUTF8String:shader_dir.c_str()];
    NSString* header = read_text([NSString stringWithUTF8String:shared_header_path.c_str()]);
    if (header == nil) {

    }

    [source appendString:@"#include <metal_stdlib>\nusing namespace metal;\n"];
    // The shared struct header is prepended so both sides agree on the layout;
    // its include guard is meaningless inside a single translation unit.
    [source appendString:[header stringByReplacingOccurrencesOfString:@"#pragma once"
                                                          withString:@""]];

    // used by pt_integrator.metal), so they are concatenated in this fixed
    // order; any other .metal files are appended afterwards alphabetically.

    NSFileManager* fm = [NSFileManager defaultManager];
    NSArray<NSString*>* names = [[fm contentsOfDirectoryAtPath:dir error:nil]
        sortedArrayUsingSelector:@selector(compare:)];
    NSMutableArray<NSString*>* ordered_names = [NSMutableArray array];
    for (NSString* name in ordered) {
        if ([names containsObject:name]) {
            [ordered_names addObject:name];
        }
    }
    for (NSString* name in names) {
        if (![name hasSuffix:@".metal"]) {
            continue;
        }
        if (![ordered containsObject:name]) {
            [ordered_names addObject:name];
        }
    }

    bool saw_kernel = false;
    for (NSString* name in ordered_names) {
        NSString* text = read_text([dir stringByAppendingPathComponent:name]);
        if (text == nil) {
            return impl_->fail(@"cannot read shader file", error);
        }
        [source appendString:@"\n"];
        [source appendString:text];
        saw_kernel = true;
    }
    if (!saw_kernel) {
        return impl_->fail(@"no .metal files found next to the binary", error);
    }

    NSError* ns_error = nil;
    MTLCompileOptions* options = [[MTLCompileOptions alloc] init];
    id<MTLLibrary> library = [impl_->device newLibraryWithSource:source options:options

        } else if (argument == "--help" || argument == "-h") {
            print_usage();
            std::exit(0);
        } else {
            std::fprintf(stderr, "unknown argument: %s\n", argument.c_str());
            return false;
        }
    }
    // The task caps host CPU work at 8 threads.
    options.threads = std::max(1, std::min(8, options.threads));
    return true;
}

std::string executable_directory(const std::string& argv0) {
    if (argv0.find('/') == std::string::npos) {
        return std::string(".");
    }
    return argv0.substr(0, argv0.find_last_of('/'));
}

double milliseconds(std::chrono::steady_clock::duration d) {
    return std::chrono::duration<double, std::milli>(d).count();
}

        return 1;

            return 1;

                    "triangles=%zu bvh_nodes=%zu\n",
                    setup_ms, samples, width, height, spp, options.threads,
                    mesh.triangles.size() / 3, bvh.nodes.size());
    }
    return 0;
}

// Reinhard tone mapping plus a 2.2 gamma encode; the linear buffer written by
// the trace kernel is copied out unchanged as the PFM, so tone mapping never
// affects any comparison.

kernel void tone_map_kernel(uint2 pixel [[thread_position_in_grid]],
                            device const PtParams& params [[buffer(0)]],
                            device const float4* radiance [[buffer(1)]],
                            device uchar4* image_out [[buffer(2)]]) {
    const uint width = params.width;
    const uint height = params.height;
    if (pixel.x >= width || pixel.y >= height) {
        return;
    }

    const uint index = pixel.y * width + pixel.x;
    float3 color = radiance[index].rgb;

    // Guard against the rare NaN/Inf from a grazing geometry term so the 8-bit
    // image is always writable; the linear buffer keeps the raw value.
    const bool invalid = any(isnan(color)) || any(isinf(color));
    color = select(color, float3(0.0f), invalid);
    color = max(color, float3(0.0f));

    const float3 mapped = color / (1.0f + color);
    const float3 encoded = pow(mapped, float3(1.0f / 2.2f));

    const uint r = uint(clamp(encoded.x, 0.0f, 1.0f) * 255.0f + 0.5f);
    const uint g = uint(clamp(encoded.y, 0.0f, 1.0f) * 255.0f + 0.5f);
    const uint b = uint(clamp(encoded.z, 0.0f, 1.0f) * 255.0f + 0.5f);
    image_out[index] = uchar4(uchar(r), uchar(g), uchar(b), uchar(255));
}
