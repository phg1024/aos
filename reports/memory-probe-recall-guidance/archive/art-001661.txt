bool quantize_bvh(const Bvh& bvh, std::vector<PtBvhNodeQ>& out, float origin[3],
                  float& inv_scale) {
    out.clear();
    if (bvh.nodes.empty()) {
        return false;
    }

    // Root box: the reference frame for the relative quantization.
    float root_min[3] = {kInfinity, kInfinity, kInfinity};
    float root_max[3] = {-kInfinity, -kInfinity, -kInfinity};
    for (const PtBvhNode& node : bvh.nodes) {
        for (int axis = 0; axis < 3; ++axis) {
            root_min[axis] = std::min(root_min[axis], node.bmin[axis]);
            root_max[axis] = std::max(root_max[axis], node.bmax[axis]);
        }
    }

    // Power-of-two scale so the decode multiply is exact; one unit step is
    // reserved as the outward rounding margin, hence 65534 rather than 65535.
    float extent = 1.0f;
    for (int axis = 0; axis < 3; ++axis) {
        extent = std::max(extent, root_max[axis] - root_min[axis]);
    }
    const double scale = std::exp2(std::floor(std::log2(double(65534.0f) / double(extent))));
    inv_scale = float(1.0 / scale);

    // Snap the origin onto the quantization grid and never above the true
    // minimum, so every quantized coordinate is non-negative.
    for (int axis = 0; axis < 3; ++axis) {
        origin[axis] = float(std::floor(double(root_min[axis]) * scale)) * inv_scale;
    }

    out.resize(bvh.nodes.size());
    for (size_t i = 0; i < bvh.nodes.size(); ++i) {
        const PtBvhNode& node = bvh.nodes[i];
        unsigned int q[3][2];
        for (int axis = 0; axis < 3; ++axis) {
            const double lo = std::floor((double(node.bmin[axis]) - double(origin[axis])) * scale);
            const double hi = std::ceil((double(node.bmax[axis]) - double(origin[axis])) * scale) +
                              1.0;
            q[axis][0] = static_cast<unsigned int>(std::max(0.0, std::min(lo, 65535.0)));
            q[axis][1] = static_cast<unsigned int>(std::max(0.0, std::min(hi, 65535.0)));
        }

        unsigned int payload = 0;
        if (node.b & PT_BVH_LEAF_BIT) {
            const unsigned int count = node.b & 0x7FFFFFFFu;
            if (count == 0 || count > 256 || node.a >= (1u << 23)) {
                return false;  // does not fit the packed leaf fields
            }
            payload = PT_BVH_LEAF_BIT | (node.a << 8) | (count - 1u);
        } else {
            if (node.a >= (1u << 31)) {
                return false;
            }
            payload = node.a;
        }

        PtBvhNodeQ& out_node = out[i];
        out_node.lo0 = static_cast<unsigned short>(q[0][0]);
        out_node.hi0 = static_cast<unsigned short>(q[0][1]);
        out_node.lo1 = static_cast<unsigned short>(q[1][0]);
        out_node.hi1 = static_cast<unsigned short>(q[1][1]);
        out_node.lo2 = static_cast<unsigned short>(q[2][0]);
        out_node.hi2 = static_cast<unsigned short>(q[2][1]);
        out_node.payload = payload;
    }
    return true;
}
