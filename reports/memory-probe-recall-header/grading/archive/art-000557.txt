    float bmax[3];
    float centroid[3];

void triangle_bounds(const Mesh& mesh, unsigned int tri_index, TriBounds& out) {
    const PtVertex* v[3] = {&mesh.vertices[mesh.triangles[3 * tri_index + 0]],
                            &mesh.vertices[mesh.triangles[3 * tri_index + 1]],
                            &mesh.vertices[mesh.triangles[3 * tri_index + 2]]};
    for (int axis = 0; axis < 3; ++axis) {
        float lo = v[0]->position[axis];
        float hi = lo;
        for (int k = 1; k < 3; ++k) {
            lo = std::min(lo, v[k]->position[axis]);
            hi = std::max(hi, v[k]->position[axis]);
        }
        out.bmin[axis] = lo;
        out.bmax[axis] = hi;
        out.centroid[axis] = 0.5f * (lo + hi);
    }
}

void union_bounds(const TriBounds& src, float bmin[3], float bmax[3]) {
    for (int axis = 0; axis < 3; ++axis) {
        bmin[axis] = std::min(bmin[axis], src.bmin[axis]);
        bmax[axis] = std::max(bmax[axis], src.bmax[axis]);
    }
}

struct BuildFrame {
    unsigned int node;
    unsigned int begin;
    unsigned int end;
};

#include <algorithm>
#include <cmath>
#include <cstdint>

namespace pt {
namespace {

constexpr unsigned int kLeafSize = 4;
constexpr unsigned int kBinCount = 16;
constexpr float kInfinity = 1e30f;
// Relative cost of visiting one internal node, in units of one triangle test.
constexpr float kTraversalCost = 0.125f;

struct TriSource {
    float bmin[3];
    float bmax[3];
    float centroid[3];
    unsigned int triangle;
};

void triangle_bounds(const Mesh& mesh, unsigned int tri_index, TriSource& out) {
    const PtVertex* v[3] = {&mesh.vertices[mesh.triangles[3 * tri_index + 0]],
                            &mesh.vertices[mesh.triangles[3 * tri_index + 1]],
                            &mesh.vertices[mesh.triangles[3 * tri_index + 2]]};
    for (int axis = 0; axis < 3; ++axis) {
        float lo = v[0]->position[axis];
        float hi = lo;
        for (int k = 1; k < 3; ++k) {
            lo = std::min(lo, v[k]->position[axis]);
            hi = std::max(hi, v[k]->position[axis]);
        }
        out.bmin[axis] = lo;
        out.bmax[axis] = hi;
        out.centroid[axis] = 0.5f * (lo + hi);
    }
}

void reset_bounds(float bmin[3], float bmax[3]) {
    for (int axis = 0; axis < 3; ++axis) {
        bmin[axis] = kInfinity;
        bmax[axis] = -kInfinity;
    }
}

void union_bounds(const float src_min[3], const float src_max[3], float bmin[3], float bmax[3]) {
    for (int axis = 0; axis < 3; ++axis) {
        bmin[axis] = std::min(bmin[axis], src_min[axis]);
        bmax[axis] = std::max(bmax[axis], src_max[axis]);
    }
}

float surface_area(const float bmin[3], const float bmax[3]) {
    const float e0 = bmax[0] - bmin[0];
    const float e1 = bmax[1] - bmin[1];
    const float e2 = bmax[2] - bmin[2];
    return 2.0f * (e0 * e1 + e1 * e2 + e2 * e0);
}

struct BinStats {
    float bmin[3];
    float bmax[3];
    unsigned int count;
};

// Top-down builder that emits nodes in pre-order (a node's left child is the
// slot created immediately after it) and packs each leaf's triangles
// contiguously into the triangle soup.
class BvhBuilder {
  public:
    BvhBuilder(const Mesh& mesh, std::vector<TriSource>& items, Bvh& out)
        : mesh_(mesh), items_(items), out_(out) {}

    unsigned int build(unsigned int begin, unsigned int end, unsigned int depth) {
        const unsigned int index = static_cast<unsigned int>(out_.nodes.size());
        out_.nodes.push_back(PtBvhNode());

        float bmin[3];
        float bmax[3];
        reset_bounds(bmin, bmax);
        for (unsigned int i = begin; i < end; ++i) {
            union_bounds(items_[i].bmin, items_[i].bmax, bmin, bmax);
        }
        for (int axis = 0; axis < 3; ++axis) {
            out_.nodes[index].bmin[axis] = bmin[axis];
            out_.nodes[index].bmax[axis] = bmax[axis];
        }

        const unsigned int count = end - begin;
        if (count <= kLeafSize || depth >= PT_BVH_MAX_DEPTH) {
            emit_leaf(index, begin, end);
            return index;
        }

        unsigned int middle = 0;
        if (!partition(begin, end, bmin, bmax, middle)) {
            emit_leaf(index, begin, end);
            return index;
        }

        out_.nodes[index].a = 0;
        out_.nodes[index].b = 0;
        build(begin, middle, depth + 1);   // lands at index + 1 (pre-order)
        const unsigned int right = build(middle, end, depth + 1);
        out_.nodes[index].a = right;
        return index;
    }

  private:
    void emit_leaf(unsigned int index, unsigned int begin, unsigned int end) {
        PtBvhNode& node = out_.nodes[index];
        node.a = static_cast<unsigned int>(out_.triangles.size());
        node.b = PT_BVH_LEAF_BIT | (end - begin);
        for (unsigned int i = begin; i < end; ++i) {
            append_triangle(items_[i].triangle);
        }
    }

    void append_triangle(unsigned int tri) {

        const unsigned int i1 = mesh_.triangles[3 * tri + 1];
        const unsigned int i2 = mesh_.triangles[3 * tri + 2];

        const float* p1 = mesh_.vertices[i1].position;
        const float* p2 = mesh_.vertices[i2].position;

            tri_out.edge1[axis] = p1[axis] - p0[axis];
            tri_out.edge2[axis] = p2[axis] - p0[axis];
        }

        const float* n2 = mesh_.vertices[i2].normal;
        for (int axis = 0; axis < 3; ++axis) {
            normal_out.n0[axis] = n0[axis];
            normal_out.n1[axis] = n1[axis];
            normal_out.n2[axis] = n2[axis];
        }
        out_.normals.push_back(normal_out);

// Splits at the median centroid along the widest axis. This is the strategy
// that produced the 105ms round-1 tree: it is always balanced, so the tree
// depth stays near log2(triangles / leaf size) and the GPU stack stays small.
// (A binned-SAH variant was tried first and regressed 40x: its unbalanced
// splits hit the depth cap and left leaves with thousands of triangles.)
    bool partition(unsigned int begin, unsigned int end, const float bmin[3], const float bmax[3],
                   unsigned int& middle) {
        float extent[3] = {bmax[0] - bmin[0], bmax[1] - bmin[1], bmax[2] - bmin[2]};
        int axis = 0;
        if (extent[1] > extent[axis]) {
            axis = 1;
        }
        if (extent[2] > extent[axis]) {
            axis = 2;
        }
        if (extent[axis] <= 0.0f) {
            return false;
        }
        const unsigned int count = end - begin;
        const unsigned int split = begin + count / 2;
        std::nth_element(items_.begin() + begin, items_.begin() + split, items_.begin() + end,
                         [axis](const TriSource& a, const TriSource& b) {
                             return a.centroid[axis] < b.centroid[axis];
                         });
        middle = split;
        return middle != begin && middle != end;
    }

    const Mesh& mesh_;
    std::vector<TriSource>& items_;
    Bvh& out_;
};

}  // namespace

        for (int axis = 0; axis < 3; ++axis) {
            v.position[axis] = scale * v.position[axis] + translate[axis];
        }
    }
}

void build_bvh(const Mesh& mesh, Bvh& out) {
    const size_t tri_count = mesh.triangles.size() / 3;
    out.nodes.clear();
    out.triangles.clear();
    out.normals.clear();
    out.nodes.reserve(tri_count);
    out.triangles.reserve(tri_count);
    out.normals.reserve(tri_count);

    std::vector<TriSource> items(tri_count);
    for (size_t i = 0; i < tri_count; ++i) {
        items[i].triangle = static_cast<unsigned int>(i);
        triangle_bounds(mesh, items[i].triangle, items[i]);
    }

    BvhBuilder builder(mesh, items, out);
    builder.build(0, static_cast<unsigned int>(tri_count), 0);
}
